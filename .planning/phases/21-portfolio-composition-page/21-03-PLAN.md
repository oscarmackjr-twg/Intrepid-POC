**Step 1 — Create `frontend/src/components/re/ConcentrationLimits.tsx` (per D-19, D-20, D-21):**

```tsx
import type { ConcentrationLimit } from '../../types/re'

interface ConcentrationLimitsProps {
  limits: ConcentrationLimit[]
}

function getBarColor(proximity: number): string {
  if (proximity >= 0.9) return 'bg-red-500'
  if (proximity >= 0.7) return 'bg-yellow-400'
  return 'bg-green-500'
}

export function ConcentrationLimits({ limits }: ConcentrationLimitsProps) {
  return (
    <div className="space-y-4">
      {limits.map((limit) => (
        <div key={limit.category}>
          <div className="flex justify-between text-sm mb-1">
            <span className="text-[#1a3868] font-medium">{limit.category}</span>
            <span className="text-[#475569]">
              {(limit.current_pct * 100).toFixed(1)}% / {(limit.limit_pct * 100).toFixed(1)}% limit
            </span>
          </div>
          <div className="w-full bg-gray-200 rounded h-2">
            <div
              className={`h-2 rounded ${getBarColor(limit.proximity)}`}
              style={{ width: `${Math.min(limit.proximity * 100, 100)}%` }}
            />
          </div>
        </div>
      ))}
    </div>
  )
}
```

Per D-19: Green (`bg-green-500`) < 70%, yellow (`bg-yellow-400`) 70-90%, red (`bg-red-500`) >= 90%.
Per D-20: `proximity` (0-1 float) drives width and color. Display: `category` name, `current_pct * 100` with 1 decimal, `limit_pct` label.
Per D-21: Outer div `w-full bg-gray-200 rounded h-2`, inner `h-2 rounded` with inline `width` style and dynamic color class. `Math.min(proximity * 100, 100)` caps width at 100% for safety.
Per D-25: TWG navy `#1a3868` for category name, slate `#475569` for percentage labels.
